Speech Separation

Model selection

Separation is a different task from the enhancement above: one microphone, two people talking at once, and one output per speaker rather than one clean output.

The model is espnet/anogkongda_librimix_enh_train_raw_valid.si_snr.ave, a Conv-TasNet trained on Libri2Mix at 8 kHz and published in the ESPnet organisation. The course used a TF-GridNet checkpoint kept on Google Drive, and a wsj0-2mix mixture from the same place; wsj0-2mix is built on WSJ0 and cannot be published, while Libri2Mix is built on LibriSpeech, which is CC BY 4.0 — so this notebook can fetch both the model and the audio it separates.

import torch
import soundfile
from espnet2.bin.enh_inference import SeparateSpeech

separate_speech = SeparateSpeech.from_pretrained(
    model_tag="espnet/anogkongda_librimix_enh_train_raw_valid.si_snr.ave",
    normalize_output_wav=True,
    device="cuda" if torch.cuda.is_available() else "cpu",
)

Separate Speech Mixture

Separate one mixture from the Libri2Mix test set

The mixture is built here rather than downloaded, the way LibriMix builds its own: take the two gains the official metadata gives for this pair, resample to the model's 8 kHz, truncate to the shorter source — LibriMix's min mode — and add them together.

Building it from the two sources is worth the six lines: it means the references are in hand, so the separation can be scored further down rather than only listened to.

# Mixture 7729-102255-0031_2094-142345-0028 of the Libri2Mix test set, named in
# metadata/Libri2Mix/libri2mix_test-clean.csv of the LibriMix repository. The
# two sources are LibriSpeech test-clean utterances (CC BY 4.0), kept beside
# the model so that this notebook does not need the corpus.
!wget -q -O spk1.flac https://huggingface.co/espnet/anogkongda_librimix_enh_train_raw_valid.si_snr.ave/resolve/main/samples/7729-102255-0031.flac
!wget -q -O spk2.flac https://huggingface.co/espnet/anogkongda_librimix_enh_train_raw_valid.si_snr.ave/resolve/main/samples/2094-142345-0028.flac

import numpy as np
import soundfile
from IPython.display import display, Audio
from scipy.signal import resample_poly

# the gains that row of the metadata gives, and the model's sampling rate
GAINS = {"spk1.flac": 0.5159350719675272, "spk2.flac": 0.22360240150798563}
FS = 8000

refs = []
for name, gain in GAINS.items():
    source, fs = soundfile.read(name, dtype="float32")
    refs.append(resample_poly(source * gain, FS, fs))
length = min(len(ref) for ref in refs)      # LibriMix's "min" mode
refs = [ref[:length] for ref in refs]
mixwav = sum(refs)
sr = FS

waves_librimix = separate_speech(mixwav[None, ...], fs=sr)

print("Input mixture", flush=True)
display(Audio(mixwav, rate=sr))
for i, wave in enumerate(waves_librimix, start=1):
    print(f"========= Separated speaker {i} =========", flush=True)
    display(Audio(wave.squeeze(), rate=sr))

How much of each speaker came back

Both sources are in hand, so this is measurable. SI-SNR asks how much of the signal is the speaker you wanted and how much is everything else, after the best scaling — one number per speaker, in dB, for the mixture going in and for the separated stream coming out.

from itertools import permutations


def si_snr(estimate, reference):
    """Scale-invariant signal-to-noise ratio, in dB."""
    estimate = estimate - estimate.mean()
    reference = reference - reference.mean()
    scale = np.dot(estimate, reference) / np.dot(reference, reference)
    target = scale * reference
    noise = estimate - target
    return 10 * np.log10(np.dot(target, target) / np.dot(noise, noise))


estimates = [wave.squeeze() for wave in waves_librimix]
# separation has no speaker order, so the scoring picks the better pairing
order = max(
    permutations(range(len(estimates))),
    key=lambda p: sum(si_snr(estimates[p[i]], refs[i]) for i in range(len(refs))),
)
estimates = [estimates[i] for i in order]

for i, (estimate, reference) in enumerate(zip(estimates, refs), start=1):
    before = si_snr(mixwav[: len(reference)], reference)
    after = si_snr(estimate[: len(reference)], reference)
    print(f"speaker {i}: {before:6.2f} dB in the mixture -> {after:6.2f} dB separated")

Show spectrums of separated speech

Show wavform and spectrogram of mixed and seperated speech.

import matplotlib.pyplot as plt
import numpy as np
import torch

from espnet2.layers.stft import Stft

stft = Stft(n_fft=512, win_length=None, hop_length=128, window="hann")
ilens = torch.LongTensor([len(mixwav)])


def magnitude(wav):
    """|STFT| of one waveform, as (frequency, time)."""
    spec = stft(torch.as_tensor(wav).reshape(1, -1), ilens)[0].squeeze()
    # the last dimension holds (real, imaginary)
    return torch.linalg.norm(spec, dim=-1).transpose(-1, -2).numpy()


def plot_spectrogram(ax, wav, title):
    mag = magnitude(wav)
    ax.imshow(
        20 * np.log10(np.maximum(mag, 1e-8)),
        origin="lower",
        aspect="auto",
        extent=[0, len(mixwav) / sr, 0, sr / 2],
        cmap="magma",
    )
    ax.set_title(title)
    ax.set_ylabel("Frequency (Hz)")


def plot_waveform(ax, wav, title):
    ax.plot(torch.linspace(0, len(mixwav) / sr, len(mixwav)), np.ravel(wav))
    ax.set_xlim(0, len(mixwav) / sr)
    ax.set_title(title)


fig, axes = plt.subplots(3, 2, figsize=(18, 9))
for row, (wav, name) in enumerate(
    [(mixwav, "Mixture"), (waves_librimix[0], "Separated (spk1)"),
     (waves_librimix[1], "Separated (spk2)")]
):
    plot_spectrogram(axes[row][0], wav, f"{name} spectrogram")
    plot_waveform(axes[row][1], wav, f"{name} waveform")
axes[2][1].set_xlabel("Time (s)")
plt.tight_layout()
plt.show()

Evaluate separated speech with a pretrained ASR model

SI-SNR says how much of each speaker came back. This asks the question the other way round: can something else use the result? The transcripts of the two sources are

text_1: HE CONTINUED HIS PRETENDED SEARCH AND TO GIVE COLOR TO HIS ERRAND MADE TWO ARRESTS

text_2: NO NO NO TOTTY UD GET HER FEET WET SAID MISSUS POYSER CARRYING AWAY HER IRON

Recognise the mixture, and the separated streams

The model is a transformer trained on LibriSpeech, which is where both of these recordings come from. It runs at 16 kHz while separation ran at 8 kHz, so each waveform is resampled on the way in — the interesting comparison is the mixture against the separated streams, and that is unaffected.

import torch
from espnet2.bin.asr_inference import Speech2Text

speech2text = Speech2Text.from_pretrained(
    model_tag=(
        "espnet/Shinji_Watanabe_librispeech_asr_train_asr_transformer_e18"
        "_raw_bpe_sp_valid.acc.best"
    ),
    ngram_weight=0.0,
    lm_weight=0.0,
    device="cuda" if torch.cuda.is_available() else "cpu",
)

ASR_FS = 16000


def recognise(wav):
    """Transcribe one 8 kHz waveform with a 16 kHz model."""
    text, *_ = speech2text(resample_poly(wav, ASR_FS, sr).astype("float32"))[0]
    return text


text_est = [recognise(estimate) for estimate in estimates]
text_m = recognise(mixwav)
print("Mix Speech to Text: ", text_m)
print("Separated Speech 1 to Text: ", text_est[0])
print("Separated Speech 2 to Text: ", text_est[1])
import difflib
from itertools import permutations

import editdistance
import numpy as np

colors = dict(
    red=lambda text: f"\033[38;2;255;0;0m{text}\033[0m" if text else "",
    green=lambda text: f"\033[38;2;0;255;0m{text}\033[0m" if text else "",
    yellow=lambda text: f"\033[38;2;225;225;0m{text}\033[0m" if text else "",
    white=lambda text: f"\033[38;2;255;255;255m{text}\033[0m" if text else "",
    black=lambda text: f"\033[38;2;0;0;0m{text}\033[0m" if text else "",
)

def diff_strings(ref, est):
    """Reference: https://stackoverflow.com/a/64404008/7384873"""
    ref_str, est_str, err_str = [], [], []
    matcher = difflib.SequenceMatcher(None, ref, est)
    for opcode, a0, a1, b0, b1 in matcher.get_opcodes():
        if opcode == "equal":
            txt = ref[a0:a1]
            ref_str.append(txt)
            est_str.append(txt)
            err_str.append(" " * (a1 - a0))
        elif opcode == "insert":
            ref_str.append("*" * (b1 - b0))
            est_str.append(colors["green"](est[b0:b1]))
            err_str.append(colors["black"]("I" * (b1 - b0)))
        elif opcode == "delete":
            ref_str.append(ref[a0:a1])
            est_str.append(colors["red"]("*" * (a1 - a0)))
            err_str.append(colors["black"]("D" * (a1 - a0)))
        elif opcode == "replace":
            diff = a1 - a0 - b1 + b0
            if diff >= 0:
                txt_ref = ref[a0:a1]
                txt_est = colors["yellow"](est[b0:b1]) + colors["red"]("*" * diff)
                txt_err = "S" * (b1 - b0) + "D" * diff
            elif diff < 0:
                txt_ref = ref[a0:a1] + "*" * -diff
                txt_est = colors["yellow"](est[b0:b1]) + colors["green"]("*" * -diff)
                txt_err = "S" * (b1 - b0) + "I" * -diff

            ref_str.append(txt_ref)
            est_str.append(txt_est)
            err_str.append(colors["black"](txt_err))
    return "".join(ref_str), "".join(est_str), "".join(err_str)


text_ref = [
  "HE CONTINUED HIS PRETENDED SEARCH AND TO GIVE COLOR TO HIS ERRAND MADE TWO ARRESTS",
  "NO NO NO TOTTY UD GET HER FEET WET SAID MISSUS POYSER CARRYING AWAY HER IRON",
]

print("=====================" , flush=True)
perms = list(permutations(range(2)))
string_edit = [
  [
    editdistance.eval(text_ref[m], text_est[n])
    for m, n in enumerate(p)
  ]
  for p in perms
]

dist = [sum(edist) for edist in string_edit]
perm_idx = np.argmin(dist)
perm = perms[perm_idx]

for i, p in enumerate(perm):
  print("\n--------------- Text %d ---------------" % (i + 1), flush=True)
  ref, est, err = diff_strings(text_ref[i], text_est[p])
  print("REF: " + ref + "\n" + "HYP: " + est + "\n" + "ERR: " + err, flush=True)
  print("Edit Distance = {}\n".format(string_edit[perm_idx][i]), flush=True)
